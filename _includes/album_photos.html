{%- comment -%} Renders every photograph in one album folder as a clickable thumbnail. Usage inside an album layout: {% include album_photos.html folder=page.folder title=page.title place=page.place %} Thumbnails live in a thumbs/ subfolder next to the originals and are generated by scripts/gallery_thumbs.py. If one is missing, the full image is used instead so a dropped-in photo still works. {%- endcomment -%} {%- assign exts = ".jpg,.jpeg,.png,.webp,.gif" -%} {%- assign thumb_prefix = include.folder | append: "thumbs/" -%} {%- assign thumb_index = "" -%} {%- for t in site.static_files -%} {%- if t.path contains thumb_prefix -%} {%- assign thumb_index = thumb_index | append: "|" | append: t.path -%} {%- endif -%} {%- endfor -%}
{%- assign n = 0 -%} {%- for f in site.static_files -%} {%- assign e = f.extname | downcase -%} {%- if exts contains e and f.path contains include.folder -%} {%- unless f.path contains "/thumbs/" -%} {%- assign n = n | plus: 1 -%} {%- assign thumb = thumb_prefix | append: f.basename | append: ".webp" -%} {%- assign needle = "|" | append: thumb -%} {%- assign src = f.path -%} {%- if thumb_index contains needle -%} {%- assign src = thumb -%} {%- endif %} {%- endunless -%} {%- endif -%} {%- endfor %}
